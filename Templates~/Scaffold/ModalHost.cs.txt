// 뷰(MainScreen/ConfirmModal)는 컨테이너를 직접 만지지 않고 이 파사드만 호출 → API 변화 흡수 지점 단일화.

// ✅ VERIFIED (2026-07-08, USN 1.7.5 실측): ns UnityScreenNavigator.Runtime.Core.Modal(단수),
//    ModalContainer.Find(string), Push(key, playAnimation, modalId=null, loadAsync=true, onLoad=null),
//    onLoad 인자 = (string modalId, Modal modal) 튜플 → x.modal 유효, Pop(playAnimation, popCount=1),
//    AsyncProcessHandle.Task(System.Threading.Tasks.Task) → .AsUniTask() 유효. 모달 프리팹 키는 Addressables.
// 복사 후 확장자를 .cs 로 변경.
// ─────────────────────────────────────────────────────────────────────────────
using Cysharp.Threading.Tasks;
using UnityScreenNavigator.Runtime.Core.Modal;

namespace Game.UI
{
    public static class ModalHost
    {
        // 프로젝트 규약: 모달 프리팹 Addressables 키.
        private const string ConfirmModalKey = "UI/ConfirmModal";

        private static ModalContainer Container => ModalContainer.Find("Main"); // ✅ Find(string) 실측 확인

        /// <summary>확인 모달을 push 하고 사용자 선택(true=확인)을 await. </summary>
        public static async UniTask<bool> PushConfirmAsync(string message)
        {
            var result = default(UniTask<bool>);

            // USN Push: 프리팹 인스턴스가 생성되면 onLoad 콜백으로 뷰를 잡아 Prepare 호출.
            var handle = Container.Push(ConfirmModalKey, playAnimation: true, onLoad: x =>
            {
                var modal = x.modal as ConfirmModal; // ✅ onLoad = (string modalId, Modal modal) 튜플
                result = modal.Prepare(message);
            });
            await handle.Task.AsUniTask(); // ✅ AsyncProcessHandle.Task → UniTask

            return await result;
        }

        /// <summary>최상단 모달 pop.</summary>
        public static void RequestPop()
        {
            Container.Pop(playAnimation: true);
        }
    }
}
