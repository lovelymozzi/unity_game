using System.Collections.Generic;
using System.Text;

namespace Game.Data
{
    /// <summary>파싱된 CSV 한 장. 헤더명으로 셀 접근.</summary>
    public sealed class CsvTable
    {
        private readonly Dictionary<string, int> _headerIndex;
        private readonly List<string[]> _rows;

        public IReadOnlyList<string> Header { get; }
        public int RowCount => _rows.Count;

        private CsvTable(List<string> header, List<string[]> rows)
        {
            Header = header;
            _rows = rows;
            _headerIndex = new Dictionary<string, int>(header.Count);
            for (int i = 0; i < header.Count; i++) _headerIndex[header[i]] = i;
        }

        /// <summary>행 인덱스 + 헤더명으로 셀 문자열. 없으면 빈 문자열.</summary>
        public string Get(int row, string column)
        {
            if (row < 0 || row >= _rows.Count) return string.Empty;
            if (!_headerIndex.TryGetValue(column, out int col)) return string.Empty;
            var cells = _rows[row];
            return col < cells.Length ? cells[col] : string.Empty;
        }

        public IEnumerable<string[]> Rows() => _rows;

        /// <summary>CSV 텍스트 전체를 파싱. 첫 줄 = 헤더.</summary>
        public static CsvTable Parse(string text)
        {
            var records = SplitRecords(text);
            if (records.Count == 0) return new CsvTable(new List<string>(), new List<string[]>());

            var header = new List<string>(records[0]);
            var rows = new List<string[]>(records.Count - 1);
            for (int i = 1; i < records.Count; i++) rows.Add(records[i]);
            return new CsvTable(header, rows);
        }

        // 상태기계 파서: 따옴표 안의 쉼표/개행을 필드 구분자로 취급하지 않음.
        private static List<string[]> SplitRecords(string text)
        {
            var records = new List<string[]>();
            var fields = new List<string>();
            var sb = new StringBuilder();
            bool inQuotes = false;
            int len = text?.Length ?? 0;

            for (int i = 0; i < len; i++)
            {
                char c = text[i];

                if (inQuotes)
                {
                    if (c == '"')
                    {
                        if (i + 1 < len && text[i + 1] == '"') { sb.Append('"'); i++; } // 이스케이프된 따옴표
                        else inQuotes = false;
                    }
                    else sb.Append(c);
                    continue;
                }

                switch (c)
                {
                    case '"':
                        inQuotes = true;
                        break;
                    case ',':
                        fields.Add(sb.ToString());
                        sb.Clear();
                        break;
                    case '\r':
                        break; // CRLF 의 CR 무시
                    case '\n':
                        fields.Add(sb.ToString());
                        sb.Clear();
                        records.Add(fields.ToArray());
                        fields.Clear();
                        break;
                    default:
                        sb.Append(c);
                        break;
                }
            }

            // 마지막 개행 없는 레코드 flush
            if (sb.Length > 0 || fields.Count > 0)
            {
                fields.Add(sb.ToString());
                records.Add(fields.ToArray());
            }
            return records;
        }
    }
}
